/* Swiper Demo 250 · flip 翻转效果 + 键盘滚轮（课堂没讲过的 NEW） */
new Swiper("#flipSwiper", {
  effect: "flip",
  grabCursor: true,
  speed: 800,
  loop: true,
  autoplay: {
    delay: 3200,
    disableOnInteraction: false
  },
  pagination: { clickable: true },
  navigation: {
    nextEl: ".swiper-button-next",
    prevEl: ".swiper-button-prev"
  },
  keyboard: { enabled: true, onlyInViewport: false },  // NEW: ⌨️ ← → 方向键翻页
  mousewheel: { sensitivity: 1 }                                       // NEW: 🖱 滚轮翻页
});
核心参数
effect: "flip"
像翻扑克牌一样沿 Y 轴 3D 翻转
专属选项
flipEffect
slideShadows:false 可去掉翻牌侧面阴影
对比 cube
flip vs cube
cube 像转动立方体；flip 像翻牌，更轻量
课堂没讲过 ⌨️
keyboard: true
← → 方向键翻页，onlyInViewport:false 全局生效
课堂没讲过 🖱
mousewheel: true
鼠标滚轮向下 = 下一页，sensitivity 可调灵敏度
